\msg_new:nnn { lralign } { too-many-alignment-tabs }
  {
    Each~side~of~an~lralign~row~may~contain~at~most~one~alignment~tab~(&).
  }

% A row without & is centred in its half of the text area.  The zero-width
% inner box is anchored at the boundary between the two quarter-width boxes.
\cs_new_protected:Npn \lralign_append_centred_side:nn #1#2
  {
    \tl_put_right:Nn \l_lralign_final_body_tl
      {
        \mathmakebox[\l_lralign_column_width_dim][r]
          { \mathmakebox[0pt][c]{ \color{#1} #2 } }
        &
        \mathmakebox[\l_lralign_column_width_dim][l]{}
      }
  }

% A row with & uses separately calculated column widths.  Their boundary is
% shifted so that the complete aligned formula group, rather than &, is
% centred in its half of the text area.
\cs_new_protected:Npn \lralign_append_aligned_side:nnnNN #1#2#3#4#5
  {
    \tl_put_right:Nn \l_lralign_final_body_tl
      {
        \mathmakebox[#4][r]
          { \color{#1} #2 }
        &
        \mathmakebox[#5][l]
          { \color{#1} {} #3 }
      }
  }

\cs_generate_variant:Nn \lralign_append_centred_side:nn { nV }
\cs_generate_variant:Nn \lralign_append_aligned_side:nnnNN { nVVNN }

% Measure the widest material before and after & just as an AMS alignment
% does.  Rows without & are centred independently and need no measurement.
\cs_new_protected:Npn \lralign_measure_row:nNN #1#2#3
  {
    \seq_set_split:Nnn \l_lralign_cell_seq { & } { #1 }
    \int_compare:nNnT { \seq_count:N \l_lralign_cell_seq } = { 2 }
      {
        \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
        \seq_pop_left:NN \l_lralign_cell_seq \l_tmpb_tl
        \hbox_set:Nn \l_lralign_measure_box
          { $\displaystyle \l_tmpa_tl$ }
        \dim_set:Nn #2
          { \dim_max:nn { #2 } { \box_wd:N \l_lralign_measure_box } }
        \hbox_set:Nn \l_lralign_measure_box
          { $\displaystyle {} \l_tmpb_tl$ }
        \dim_set:Nn #3
          { \dim_max:nn { #3 } { \box_wd:N \l_lralign_measure_box } }
      }
  }

\cs_new_protected:Npn \lralign_measure_rows:NNN #1#2#3
  {
    \dim_zero:N #2
    \dim_zero:N #3
    \seq_map_inline:Nn #1
      { \lralign_measure_row:nNN { ##1 } #2 #3 }
  }

\cs_new_protected:Npn \lralign_append_side:nnNN #1#2#3#4
  {
    \seq_set_split:Nnn \l_lralign_cell_seq { & } { #2 }
    \int_case:nnF { \seq_count:N \l_lralign_cell_seq }
      {
        { 1 }
          {
            \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
            \lralign_append_centred_side:nV { #1 } \l_tmpa_tl
          }
        { 2 }
          {
            \seq_pop_left:NN \l_lralign_cell_seq \l_tmpa_tl
            \seq_pop_left:NN \l_lralign_cell_seq \l_tmpb_tl
            \lralign_append_aligned_side:nVVNN
              { #1 } \l_tmpa_tl \l_tmpb_tl #3 #4
          }
      }
      { \msg_error:nn { lralign } { too-many-alignment-tabs } }
  }

\cs_generate_variant:Nn \lralign_append_side:nnNN { nVNN }

\NewDocumentEnvironment{lralign}{ +b }{
    % Split input at \switch
    \seq_set_split:Nnn \l_lralign_parts_seq { \switch } { #1 }
    
    % Extract Left and Right
    \seq_pop_left:NN \l_lralign_parts_seq \l_lralign_left_body_tl
    % Check if Right exists
    \seq_if_empty:NTF \l_lralign_parts_seq
    { \tl_clear:N \l_lralign_right_body_tl }
    { \seq_pop_left:NN \l_lralign_parts_seq \l_lralign_right_body_tl }
    
    % Split into rows (using \\ as delimiter)
    \seq_set_split:NnV \l_lralign_left_rows_seq { \\ } \l_lralign_left_body_tl
    \seq_set_split:NnV \l_lralign_right_rows_seq { \\ } \l_lralign_right_body_tl
    
    % Calculate max rows
    \int_set:Nn \l_lralign_max_rows_int { \int_max:nn { \seq_count:N \l_lralign_left_rows_seq } { \seq_count:N \l_lralign_right_rows_seq } }
    
    \tl_clear:N \l_lralign_final_body_tl
    \dim_set:Nn \l_lralign_column_width_dim { .25\linewidth }

    % Find the natural width of both alignment columns on each side.
    \lralign_measure_rows:NNN
      \l_lralign_left_rows_seq
      \l_lralign_left_pre_max_dim
      \l_lralign_left_post_max_dim
    \lralign_measure_rows:NNN
      \l_lralign_right_rows_seq
      \l_lralign_right_pre_max_dim
      \l_lralign_right_post_max_dim

    % Centre each complete aligned group in its half.  Clamp the alignment
    % point to that half for exceptionally wide or strongly asymmetric rows.
    \dim_set:Nn \l_lralign_left_pre_column_dim
      {
        \dim_max:nn { 0pt }
          {
            \dim_min:nn { .5\linewidth }
              {
                .25\linewidth
                + ( \l_lralign_left_pre_max_dim
                    - \l_lralign_left_post_max_dim ) / 2
              }
          }
      }
    \dim_set:Nn \l_lralign_left_post_column_dim
      { .5\linewidth - \l_lralign_left_pre_column_dim }
    \dim_set:Nn \l_lralign_right_pre_column_dim
      {
        \dim_max:nn { 0pt }
          {
            \dim_min:nn { .5\linewidth }
              {
                .25\linewidth
                + ( \l_lralign_right_pre_max_dim
                    - \l_lralign_right_post_max_dim ) / 2
              }
          }
      }
    \dim_set:Nn \l_lralign_right_post_column_dim
      { .5\linewidth - \l_lralign_right_pre_column_dim }
    
    % Loop and construct alignment body
    \int_step_inline:nn { \l_lralign_max_rows_int } {
        % Process left side
        \seq_pop_left:NN \l_lralign_left_rows_seq \l_tmpa_tl
        \quark_if_no_value:NTF \l_tmpa_tl
        { \tl_clear:N \l_tmpa_tl }
        { }
        \lralign_append_side:nVNN
          { blue } \l_tmpa_tl
          \l_lralign_left_pre_column_dim
          \l_lralign_left_post_column_dim
        
        % Alignment tab between the two fixed-width sides
        \tl_put_right:Nn \l_lralign_final_body_tl { & }
        
        % Process right side
        \seq_pop_left:NN \l_lralign_right_rows_seq \l_tmpa_tl
        \quark_if_no_value:NTF \l_tmpa_tl
        { \tl_clear:N \l_tmpa_tl }
        { }
        \lralign_append_side:nVNN
          { red } \l_tmpa_tl
          \l_lralign_right_pre_column_dim
          \l_lralign_right_post_column_dim
        
        % Append \\ unless it's the last row
        \int_compare:nNnF { ##1 } = { \l_lralign_max_rows_int } {
            \tl_put_right:Nn \l_lralign_final_body_tl { \\ }
        }
    }
    
    % alignat* remains a top-level AMS display, so \allowdisplaybreaks applies.
    \begin{alignat*}{2}
    \l_lralign_final_body_tl
    \end{alignat*}
}{\ignorespacesafterend}
